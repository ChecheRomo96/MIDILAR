84 void Process(const Protocol::Packet& packet) {
85 const Protocol::NoteNumber note = packet.Note();
86 if (!note.IsValid()) {
87 Emit(packet);
88 return;
89 }
90 const Protocol::NoteNumber mapped = Protocol::NoteNumber::FromValue(_held.Route(packet, Map(note).Value()));
91 if (mapped.IsValid()) {
92 Emit(packet.WithNote(mapped));
93 }
94 }

97 Protocol::NoteNumber Map(Protocol::NoteNumber note) const noexcept {
98 if (!note.IsValid() || !_scale.IsValid() || InScale(note.Value())) {
99 return note;
100 }
101 if (_mode == ScaleFilterMode::Drop) {
102 return Protocol::NoteNumber::Invalid();
103 }
104 for (int32_t distance = 1; distance < 12; ++distance) {
105 const int32_t below = note.Value() - distance;
106 const int32_t above = note.Value() + distance;
107 if (_mode != ScaleFilterMode::Up && InScale(below)) {
108 return Protocol::NoteNumber::FromValue(below);
109 }
110 if (_mode != ScaleFilterMode::Down && InScale(above)) {
111 return Protocol::NoteNumber::FromValue(above);
112 }
113 }
114 return Protocol::NoteNumber::Invalid();
115 }
116
117private:
118 bool InScale(int32_t value) const noexcept {
119 const Protocol::NoteNumber note = Protocol::NoteNumber::FromValue(value);
120 return note.IsValid() && _scale.ContainsPitchClass(Protocol::ToChromaticIndex(note).PitchClass());
121 }
122
123 HeldNotes _held;
124 MCC::Scale _scale;
125 ScaleFilterMode _mode = ScaleFilterMode::Nearest;
126};

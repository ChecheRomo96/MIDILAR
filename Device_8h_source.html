74 const bool noteOn = status == Protocol::VoiceStatus::NoteOn &&
75 !(packet.Type() == Protocol::MessageType::Midi1ChannelVoice && (packet.Word(0) & 0x7Fu) == 0);

77 (status == Protocol::VoiceStatus::NoteOn && !noteOn);

94 }
95 return stored;
96 }

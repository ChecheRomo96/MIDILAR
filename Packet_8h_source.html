307 static FOUNDATION_CONSTEXPR14 Packet SysEx7(Protocol::Ump::Group group, Protocol::Ump::SysExStatus status, const uint8_t* data,
308 uint8_t size) noexcept {
309 if (!group.IsValid() || size > 6 || (data == nullptr && size > 0)) {
310 return Packet();
311 }
312 uint32_t bytes[6] = {0, 0, 0, 0, 0, 0};
313 for (uint8_t i = 0; i < size; ++i) {
314 if (data[i] > 127) {
315 return Packet();
316 }
317 bytes[i] = data[i];
318 }
319 return Packet(Header(MessageType::Data64, group.Wire(), static_cast<uint32_t>(status)) |
320 (static_cast<uint32_t>(size) << 16) | (bytes[0] << 8) | bytes[1],
321 (bytes[2] << 24) | (bytes[3] << 16) | (bytes[4] << 8) | bytes[5], 0, 0);
322 }

368 static FOUNDATION_CONSTEXPR14 Packet SysEx7(Protocol::Group group, Protocol::SysExStatus status, const uint8_t* data,
369 uint8_t size) noexcept {
370 if (!group.IsValid() || size > 6 || (data == nullptr && size > 0)) {
371 return Packet();
372 }
373 uint32_t bytes[6] = {0, 0, 0, 0, 0, 0};
374 for (uint8_t i = 0; i < size; ++i) {
375 if (data[i] > 127) {
376 return Packet();
377 }
378 bytes[i] = data[i];
379 }
380 return Packet(Header(MessageType::Data64, group.Wire(), static_cast<uint32_t>(status)) |
381 (static_cast<uint32_t>(size) << 16) | (bytes[0] << 8) | bytes[1],
382 (bytes[2] << 24) | (bytes[3] << 16) | (bytes[4] << 8) | bytes[5], 0, 0);
383 }

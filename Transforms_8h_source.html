98class VelocityCurve : public Device {
99public:
101 VelocityMap& Map() noexcept { return _map; }
102
104 void Process(const Protocol::Packet& packet) const {
105 if (!_map.IsBound() || !packet.IsChannelVoice() ||
106 packet.VoiceStatus() != Protocol::VoiceStatus::NoteOn) {
107 Emit(packet);
108 return;
109 }
110 const Protocol::Velocity velocity = _map.Invoke(packet.Velocity());
111 if (packet.Type() == Protocol::MessageType::Midi1ChannelVoice) {
112 Emit(Protocol::Packet::Midi1NoteOn(packet.Group(), packet.Channel(), packet.Note(), velocity));
113 } else {
114 Emit(Protocol::Packet::Midi2NoteOn(packet.Group(), packet.Channel(), packet.Note(), velocity,
115 packet.AttributeType(), packet.AttributeData()));
116 }

134class ChordGenerator : public Device {
135public:
137 void SetChord(const MCC::ChordPattern& chord) noexcept {
138 _shape = 1;
139 if (!chord.IsValid()) {
140 return;
141 }
142 _shape = 0;
143 for (int32_t tone = 1; tone <= chord.ToneCount(); ++tone) {
144 const int32_t semitones = chord.ToneInterval(tone).Semitones();
145 if (semitones >= 0 && semitones < 32) {
146 _shape |= static_cast<uint32_t>(1) << semitones;
147 }
148 }
149 }
150
152 uint32_t Shape() const noexcept { return _shape; }
153
155 void Process(const Protocol::Packet& packet) {
156 const Protocol::NoteNumber note = packet.Note();
157 if (!note.IsValid()) {
158 Emit(packet);
159 return;
160 }
161 const uint32_t shape = _held.Route(packet, _shape);
162 for (uint8_t semitones = 0; semitones < 32; ++semitones) {
163 if (((shape >> semitones) & 1u) != 0) {
164 const Protocol::NoteNumber tone = Protocol::NoteNumber::FromValue(note.Value() + semitones);
165 if (tone.IsValid()) {
166 Emit(packet.WithNote(tone));
167 }
168 }
169 }
170 }
171
172private:
173 HeldValues<uint32_t> _held;
174 uint32_t _shape = 1;
175};

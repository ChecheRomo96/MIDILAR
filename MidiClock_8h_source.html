90 void Update() {
91 const uint32_t now = _clock.Now().Ticks();
92 if (!_started) {
93 _started = true;
94 _next = now;
95 }
96 uint8_t sent = 0;
97 while (static_cast<int32_t>(now - _next) >= 0) {
98 if (sent == MaxCatchUp) {
99 // Too far behind: drop the backlog and restart the schedule.
100 _next = now;
101 _remainder = 0;
102 Advance();
103 return;
104 }
105 Emit(Protocol::Packet::System(_group, Protocol::SystemStatus::TimingClock));
106 if (_running) {
107 ++_position;
108 }
109 ++sent;
110 Advance();
111 }
112 }
113
115 bool IsRunning() const noexcept { return _running; }
116
118 uint32_t Position() const noexcept { return _position; }
119
120private:
121 void Advance() noexcept {
122 const uint64_t total = _stepNumerator + _remainder;
123 const uint64_t step = _stepDenominator == 0 ? 1 : total / _stepDenominator;
124 _remainder = _stepDenominator == 0 ? 0 : total % _stepDenominator;
125 _next += static_cast<uint32_t>(step == 0 ? 1 : step);
126 }
127
128 const Foundation::Time::Clock& _clock;
129 Protocol::Group _group;
130 uint32_t _tempo = 0;
131 uint64_t _stepNumerator = 0;
132 uint64_t _stepDenominator = 0;
133 uint64_t _remainder = 0;
134 uint32_t _next = 0;
135 uint32_t _position = 0;
136 bool _started = false;
137 bool _running = false;

150class ClockReceiver : public Device {
151public:
153 explicit ClockReceiver(const Foundation::Time::Clock& clock) noexcept : _clock(clock) {}
154
156 void Process(const Protocol::Packet& packet) {
157 if (packet.Type() == Protocol::MessageType::System) {
158 switch (packet.SystemStatus()) {
159 case Protocol::SystemStatus::TimingClock:
160 Tick();
161 break;
162 case Protocol::SystemStatus::Start:
163 _running = true;
164 _position = 0;
165 break;
166 case Protocol::SystemStatus::Continue:
167 _running = true;
168 break;
169 case Protocol::SystemStatus::Stop:
170 _running = false;
171 break;
172 case Protocol::SystemStatus::SongPosition:
173 if (!_running) {
174 _position = static_cast<uint32_t>(packet.SongPosition()) * ClocksPerSongPositionBeat;
175 }
176 break;
177 default:
178 break;
179 }
180 }
181 Emit(packet);
182 }
183
185 bool IsRunning() const noexcept { return _running; }
186
188 uint32_t Position() const noexcept { return _position; }
189
191 uint32_t Tempo() const noexcept {
192 if (_interval == 0) {
193 return 0;
194 }
195 const Foundation::Time::Frequency frequency = _clock.GetFrequency();
196 // centiBpm = frequency * 60 * 100 / (24 * interval), interval in ticks * 8.
197 const uint64_t numerator = static_cast<uint64_t>(frequency.Numerator()) * 250u * 8u;
198 const uint64_t denominator = static_cast<uint64_t>(frequency.Denominator()) * _interval;
199 return denominator == 0 ? 0 : static_cast<uint32_t>((numerator + denominator / 2) / denominator);
200 }
201
202private:
203 void Tick() {
204 const uint32_t now = _clock.Now().Ticks();
205 if (_hasLast) {
206 const uint32_t interval = now - _last;
207 // Exponential average kept as 8 x interval for precision.
208 _interval = _interval == 0 ? interval * 8u : _interval - _interval / 8u + interval;
209 }
210 _last = now;
211 _hasLast = true;
212 if (_running) {
213 ++_position;
214 }
215 }
216
217 const Foundation::Time::Clock& _clock;
218 uint32_t _last = 0;
219 uint32_t _interval = 0;
220 uint32_t _position = 0;
221 bool _hasLast = false;
222 bool _running = false;
223};

28constexpr uint64_t EuclideanPattern(uint8_t pulses, uint8_t steps, uint8_t rotation = 0) noexcept {
29 if (steps == 0 || steps > MaxSteps) {
30 return 0;
31 }
32 const uint32_t onsets = pulses > steps ? steps : pulses;
33 uint64_t pattern = 0;
34 for (uint32_t i = 0; i < steps; ++i) {
35 if ((i * onsets) % steps < onsets) {
36 pattern |= static_cast<uint64_t>(1) << ((i + rotation) % steps);
37 }
38 }
39 return pattern;
40}

106 void Process(const Protocol::Packet& packet) {
107 Emit(packet);
108 if (packet.Type() != Protocol::MessageType::System) {
109 return;
110 }
111 switch (packet.SystemStatus()) {
112 case Protocol::SystemStatus::Start:
113 Release();
114 _running = true;
115 _clock = 0;
116 break;
117 case Protocol::SystemStatus::Continue:
118 _running = true;
119 break;
120 case Protocol::SystemStatus::Stop:
121 _running = false;
122 Release();
123 break;
124 case Protocol::SystemStatus::SongPosition:
125 if (!_running) {
126 _clock = static_cast<uint32_t>(packet.SongPosition()) * ClocksPerSongPositionBeat;
127 }
128 break;
129 case Protocol::SystemStatus::TimingClock:
130 if (_running) {
131 Tick();
132 }
133 break;
134 default:
135 break;
136 }
137 }

139private:
140 void Tick() {
141 if (_sounding && --_gateLeft == 0) {
142 Release();
143 }
144 if (_clock % _clocksPerStep == 0) {
145 const uint8_t step = CurrentStep();
146 if (((_pattern >> step) & 1u) != 0) {
147 Release();
148 _playing = _note;
149 _sounding = true;
150 _gateLeft = _gate;
151 Emit(Protocol::Packet::Midi2NoteOn(_group, _channel, _playing, _velocity));
152 }
153 }
154 ++_clock;
155 }

157 void Release() {
158 if (_sounding) {
159 _sounding = false;
160 Emit(Protocol::Packet::Midi2NoteOff(_group, _channel, _playing, Protocol::Velocity()));
161 }
162 }

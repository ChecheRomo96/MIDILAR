44 static constexpr uint8_t Length(uint8_t status) noexcept {
45 return status >= 0x80 && status < 0xF0
46 ? ((status & 0xF0u) == 0xC0 || (status & 0xF0u) == 0xD0 ? 2 : 3)
47 : status == 0xF2 ? 3
48 : status == 0xF1 || status == 0xF3 ? 2
49 : status == 0xF6 || status == 0xF8 || status == 0xFA ||
50 status == 0xFB || status == 0xFC || status == 0xFE || status == 0xFF ? 1 : 0;
51 }

53 Message& Set(uint8_t status, uint8_t first = 0, uint8_t second = 0) noexcept {
54 _bytes[0] = status;
55 _bytes[1] = first;
56 _bytes[2] = second;
57 return *this;
58 }

60 Message& Voice(uint8_t status, int32_t first, int32_t second, int32_t channel) noexcept {
61 if (channel < 0 || channel > 15 || first < 0 || first > 127 || second < 0 || second > 127)
62 return Clear();
63 return Set(static_cast<uint8_t>(status | channel),
64 static_cast<uint8_t>(first), static_cast<uint8_t>(second));
65 }

97 Message& SetRawData(const uint8_t* bytes, size_t count) noexcept {
98 if (bytes == nullptr || count == 0 || count > 3 || Length(bytes[0]) != count)
99 return Clear();
100 const uint8_t first = count > 1 ? bytes[1] : 0;
101 const uint8_t second = count > 2 ? bytes[2] : 0;
102 if (first > 127 || second > 127) return Clear();
103 return Set(bytes[0], first, second);
104 }

44 uint8_t Encode(const Packet& packet, uint8_t (&out)[MaxBytes]) noexcept {
45 if (!packet.IsValid()) {
46 return 0;
47 }
48 switch (packet.Type()) {
49 case MessageType::System:
50 return EncodeSystem(packet, out);
51 case MessageType::Midi1ChannelVoice:
52 return EncodeVoice(packet.Word(0), out, 0);
53 case MessageType::Midi2ChannelVoice:
54 return EncodeMidi2(packet, out);
55 case MessageType::Data64:
56 return EncodeSysEx(packet, out);

63 uint8_t EncodeSystem(const Packet& packet, uint8_t (&out)[MaxBytes]) noexcept {
64 const uint8_t status = static_cast<uint8_t>(packet.SystemStatus());
65 if (status < 0xF1 || status == 0xF4 || status == 0xF5 || status == 0xF7 || status == 0xF9 ||
66 status == 0xFD) {
67 return 0;
68 }
69 out[0] = status;
70 if (status >= 0xF8) {
71 return 1;
72 }
73 _runningStatus = 0;
74 if (status == 0xF6) {
75 return 1;
76 }
77 out[1] = static_cast<uint8_t>((packet.Word(0) >> 8) & 0x7Fu);
78 if (status != 0xF2) {
79 return 2;
80 }
81 out[2] = static_cast<uint8_t>(packet.Word(0) & 0x7Fu);
82 return 3;
83 }

85 uint8_t EncodeVoice(uint32_t word, uint8_t (&out)[MaxBytes], uint8_t count) noexcept {
86 const uint8_t status = static_cast<uint8_t>((word >> 16) & 0xFFu);
87 if (status < 0x80 || status >= 0xF0) {
88 return count;
89 }
90 if (!_useRunningStatus || status != _runningStatus) {
91 out[count++] = status;
92 }
93 _runningStatus = status;
94 out[count++] = static_cast<uint8_t>((word >> 8) & 0x7Fu);
95 if ((status & 0xF0) != 0xC0 && (status & 0xF0) != 0xD0) {
96 out[count++] = static_cast<uint8_t>(word & 0x7Fu);
97 }
98 return count;
99 }

101 uint8_t EncodeMidi2(const Packet& packet, uint8_t (&out)[MaxBytes]) noexcept {
102 const Group group = packet.Group();
103 const Channel channel = packet.Channel();
104 switch (packet.VoiceStatus()) {
105 case VoiceStatus::NoteOn:
106 return EncodeVoice(Packet::Midi1NoteOn(group, channel, packet.Note(), packet.Velocity()).Word(0), out, 0);
107 case VoiceStatus::NoteOff:
108 return EncodeVoice(Packet::Midi1NoteOff(group, channel, packet.Note(), packet.Velocity()).Word(0), out,
109 0);
110 case VoiceStatus::PolyPressure:
111 return EncodeVoice(Packet::Midi1PolyPressure(group, channel, packet.Note(), packet.Pressure()).Word(0),
112 out, 0);
113 case VoiceStatus::ControlChange:
114 return EncodeVoice(
115 Packet::Midi1ControlChange(group, channel, packet.Controller(), packet.ControllerValue()).Word(0),
116 out, 0);
117 case VoiceStatus::ChannelPressure:
118 return EncodeVoice(Packet::Midi1ChannelPressure(group, channel, packet.Pressure()).Word(0), out, 0);
119 case VoiceStatus::PitchBendChange:
120 return EncodeVoice(Packet::Midi1PitchBend(group, channel, packet.PitchBend()).Word(0), out, 0);
121 case VoiceStatus::ProgramChange: {
122 uint8_t count = 0;
123 if (packet.HasBank()) {
124 count = EncodeVoice(Packet::Midi1ControlChange(group, channel, ControllerNumber::FromValue(0),
125 ControllerValue::FromMidi1(packet.Bank() >> 7))
126 .Word(0),
127 out, count);
128 count = EncodeVoice(Packet::Midi1ControlChange(group, channel, ControllerNumber::FromValue(32),
129 ControllerValue::FromMidi1(packet.Bank() & 0x7F))
130 .Word(0),
131 out, count);
132 }
133 return EncodeVoice(Packet::Midi1ProgramChange(group, channel, packet.Program()).Word(0), out, count);
134 }
135 default:
136 return 0;
137 }

140 uint8_t EncodeSysEx(const Packet& packet, uint8_t (&out)[MaxBytes]) noexcept {
141 const SysExStatus status = packet.SysExStatus();
142 if (static_cast<uint8_t>(status) > static_cast<uint8_t>(SysExStatus::End) ||
143 ((packet.Word(0) >> 16) & 0xFu) > 6) {
144 return 0;
145 }
146 _runningStatus = 0;
147 uint8_t count = 0;
148 if (status == SysExStatus::Complete || status == SysExStatus::Start) {
149 out[count++] = 0xF0;
150 }
151 for (uint8_t i = 0; i < packet.SysExSize(); ++i) {
152 out[count++] = static_cast<uint8_t>(packet.SysExByte(i) & 0x7Fu);
153 }
154 if (status == SysExStatus::Complete || status == SysExStatus::End) {
155 out[count++] = 0xF7;
156 }
157 return count;
158 }

157 static FOUNDATION_CONSTEXPR14 Message FromDataWords(uint32_t header, uint32_t data) noexcept {
158 if ((header & 0xFF000000u) != 0) return Message();
159 const Protocol::VoiceStatus status = static_cast<Protocol::VoiceStatus>((header >> 20) & 15u);
160 switch (status) {
161 case Protocol::VoiceStatus::NoteOn:
162 case Protocol::VoiceStatus::NoteOff:
163 if ((header & 0x8000u) != 0 || ((header & 255u) == 0 && (data & 0xFFFFu) != 0)) return Message();
164 break;
165 case Protocol::VoiceStatus::PolyPressure:
166 case Protocol::VoiceStatus::ControlChange:
167 if ((header & 0x80FFu) != 0) return Message();
168 break;
169 case Protocol::VoiceStatus::ProgramChange:
170 if ((header & 0xFFFEu) != 0 || (data & 0x80FF8080u) != 0 ||
171 ((header & 1u) == 0 && (data & 0xFFFFu) != 0)) return Message();
172 break;
173 case Protocol::VoiceStatus::ChannelPressure:
174 case Protocol::VoiceStatus::PitchBendChange:
175 if ((header & 0xFFFFu) != 0) return Message();
176 break;
177 default: return Message();
178 }
179 return Message(header, data);
180 }

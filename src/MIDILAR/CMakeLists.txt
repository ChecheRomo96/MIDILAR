set(MIDILAR_MODULE_HEADERS "")

if(MIDILAR_CORE)
    list(APPEND MIDILAR_MODULE_HEADERS "${CMAKE_CURRENT_LIST_DIR}/Core.h")
endif()

target_sources(MIDILAR PRIVATE ${MIDILAR_MODULE_HEADERS})

install(FILES ${MIDILAR_MODULE_HEADERS}
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/MIDILAR"
    CONFIGURATIONS Release)

    add_subdirectory(Core)
endif()

27inline bool DecodeUsbMidi1(const uint8_t (&event)[4], Packet& out) noexcept {
28 const Group group = Group::FromWire(event[0] >> 4);
29 const uint8_t cin = static_cast<uint8_t>(event[0] & 0x0Fu);
30 const uint8_t b1 = event[1];
31 const uint8_t b2 = event[2];
32 const uint8_t b3 = event[3];
33
34 if (cin == 0xF) {
35 // Single byte: real-time messages.
36 if (b1 < 0xF8 || b1 == 0xF9 || b1 == 0xFD) {
37 return false;
38 }
39 out = Packet::System(group, static_cast<SystemStatus>(b1));
40 return true;
41 }
42
43 if (cin >= 0x8) {
44 // Channel voice: the CIN repeats the status nibble.
45 if ((b1 >> 4) != cin || b2 > 127 || b3 > 127) {
46 return false;
47 }
48 const bool oneData = cin == 0xC || cin == 0xD;
49 const bool noteOnAsOff = cin == 0x9 && b3 == 0;
50 const uint32_t status = noteOnAsOff ? (0x80u | (b1 & 0x0Fu)) : b1;
51 out = Packet::FromWords((static_cast<uint32_t>(0x2) << 28) | (static_cast<uint32_t>(group.Wire()) << 24) |
52 (status << 16) | (static_cast<uint32_t>(b2) << 8) | (oneData ? 0u : b3));
53 return true;
54 }
55
56 if (cin == 0x2 || cin == 0x3) {
57 // Two- and three-byte system common.
58 if (cin == 0x2 && b1 == 0xF1) {
59 out = Packet::TimeCode(group, b2);
60 } else if (cin == 0x2 && b1 == 0xF3) {
61 out = Packet::SongSelect(group, b2);
62 } else if (cin == 0x3 && b1 == 0xF2 && b2 <= 127 && b3 <= 127) {
63 out = Packet::SongPosition(group, static_cast<uint16_t>((b3 << 7) | b2));
64 } else {
65 return false;
66 }
67 return out.IsValid();
68 }
69
70 if (cin == 0x5 && b1 == 0xF6) {
71 out = Packet::System(group, SystemStatus::TuneRequest);
72 return true;
73 }
74
75 if (cin >= 0x4 && cin <= 0x7) {
76 // System Exclusive: 4 = start or continue (3 bytes), 5/6/7 = ends
77 // with 1/2/3 bytes.
78 const uint8_t size = cin == 0x4 ? 3 : static_cast<uint8_t>(cin - 0x4);
79 const uint8_t bytes[3] = {b1, b2, b3};
80 const bool start = bytes[0] == 0xF0;
81 const bool end = cin != 0x4 && bytes[size - 1] == 0xF7;
82 if (cin != 0x4 && !end) {
83 return false;
84 }
85 const uint8_t first = start ? 1 : 0;
86 const uint8_t count = static_cast<uint8_t>(size - first - (end ? 1 : 0));
87 const SysExStatus status = start ? (end ? SysExStatus::Complete : SysExStatus::Start)
88 : (end ? SysExStatus::End : SysExStatus::Continue);
89 out = Packet::SysEx7(group, status, bytes + first, count);
90 return out.IsValid();
91 }
92
93 return false;
94}

120 uint8_t Encode(const Packet& packet, Event (&out)[MaxEvents]) noexcept {
121 const Group group = packet.Group();
122 if (!group.IsValid()) {
123 return 0;
124 }
125 uint8_t bytes[Midi1Encoder::MaxBytes];
126 const uint8_t size = _encoder.Encode(packet, bytes);
127 Cable& cable = _cables[group.Wire()];
128 const uint8_t prefix = static_cast<uint8_t>(group.Wire() << 4);
129 uint8_t count = 0;
130 uint8_t index = 0;
131 while (index < size) {
132 const uint8_t byte = bytes[index];
133 if (byte >= 0xF8) {
134 Put(out[count++], static_cast<uint8_t>(prefix | 0xF), byte, 0, 0);
135 ++index;
136 } else if (byte == 0xF0 || cable.inSysEx) {
137 if (byte == 0xF0) {
138 cable.inSysEx = true;
139 cable.count = 0;
140 }
141 cable.pending[cable.count++] = byte;
142 if (byte == 0xF7) {
143 Put(out[count++], static_cast<uint8_t>(prefix | (0x4 + cable.count)), cable.pending[0],
144 cable.count > 1 ? cable.pending[1] : 0, cable.count > 2 ? cable.pending[2] : 0);
145 cable.inSysEx = false;
146 cable.count = 0;
147 } else if (cable.count == 3) {
148 Put(out[count++], static_cast<uint8_t>(prefix | 0x4), cable.pending[0], cable.pending[1],
149 cable.pending[2]);
150 cable.count = 0;
151 }
152 ++index;
153 } else if (byte < 0x80) {
154 // System Exclusive data without a start on this cable.
155 ++index;
156 } else {
157 const uint8_t length = MessageLength(byte);
158 const uint8_t cin = byte < 0xF0 ? static_cast<uint8_t>(byte >> 4)
159 : byte == 0xF2 ? 0x3
160 : byte == 0xF6 ? 0x5
161 : 0x2;
162 Put(out[count++], static_cast<uint8_t>(prefix | cin), byte,
163 length > 1 ? bytes[index + 1] : 0, length > 2 ? bytes[index + 2] : 0);
164 index = static_cast<uint8_t>(index + length);
165 }
166 }
167 return count;
168 }
169
170private:
171 struct Cable {
172 uint8_t pending[3];
173 uint8_t count;
174 bool inSysEx;
175 };
176
177 static uint8_t MessageLength(uint8_t status) noexcept {
178 return (status & 0xF0) == 0xC0 || (status & 0xF0) == 0xD0 || status == 0xF1 || status == 0xF3 ? 2
179 : status == 0xF6 ? 1
180 : 3;
181 }
182
183 static void Put(Event& event, uint8_t header, uint8_t b1, uint8_t b2, uint8_t b3) noexcept {
184 event[0] = header;
185 event[1] = b1;
186 event[2] = b2;
187 event[3] = b3;
188 }
189
190 Midi1Encoder _encoder;
191 Cable _cables[16] = {};
192};

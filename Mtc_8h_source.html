43 void Locate(Protocol::TimeCode time, bool announce = true) {
44 if (!time.IsValid()) {
45 return;
46 }
47 _time = time;
48 _piece = 0;
49 // Ticks per quarter frame = frequency / (fps * 4).
50 const Foundation::Time::Frequency frequency = _clock.GetFrequency();
51 const bool ntsc = time.Rate() == Protocol::TimeCodeRate::Fps29_97Drop;
52 _stepNumerator = static_cast<uint64_t>(frequency.Numerator()) * (ntsc ? 1001u : 1u);
53 _stepDenominator = static_cast<uint64_t>(frequency.Denominator()) *
54 (ntsc ? 30000u : Protocol::TimeCode::FramesPerSecond(time.Rate())) * 4u;
55 _remainder = 0;
56 if (announce) {
57 SendFullFrame();
58 }
59 }
60
62 void Start() {
63 _running = true;
64 _piece = 0;
65 _remainder = 0;
66 _next = _clock.Now().Ticks();
67 }
68
70 void Stop() noexcept { _running = false; }
71
73 bool IsRunning() const noexcept { return _running; }
74
76 Protocol::TimeCode Time() const noexcept { return _time; }
77
79 void SendFullFrame() {
80 const uint8_t head[6] = {0x7F, 0x7F, 0x01, 0x01, _time.RateAndHours(), _time.Minutes()};
81 const uint8_t tail[2] = {_time.Seconds(), _time.Frames()};
82 Emit(Protocol::Packet::SysEx7(_group, Protocol::SysExStatus::Start, head, 6));
83 Emit(Protocol::Packet::SysEx7(_group, Protocol::SysExStatus::End, tail, 2));
84 }
85
87 void Update() {
88 if (!_running) {
89 return;
90 }
91 const uint32_t now = _clock.Now().Ticks();
92 uint8_t sent = 0;
93 while (static_cast<int32_t>(now - _next) >= 0) {
94 if (sent == MaxCatchUp) {
95 _next = now;
96 _remainder = 0;
97 Advance();
98 return;
99 }
100 if (_piece == 0) {
101 _message = _time;
102 }
103 Emit(Protocol::Packet::TimeCode(_group, _message.QuarterFrame(_piece)));
104 _piece = static_cast<uint8_t>((_piece + 1) & 7u);
105 if ((_piece & 3u) == 0) {
106 _time = _time.Next();
107 }
108 ++sent;
109 Advance();
110 }
111 }
112
113private:
114 void Advance() noexcept {
115 const uint64_t total = _stepNumerator + _remainder;
116 const uint64_t step = _stepDenominator == 0 ? 1 : total / _stepDenominator;
117 _remainder = _stepDenominator == 0 ? 0 : total % _stepDenominator;
118 _next += static_cast<uint32_t>(step == 0 ? 1 : step);
119 }
120
121 const Foundation::Time::Clock& _clock;
122 Protocol::Group _group;
123 Protocol::TimeCode _time;
124 Protocol::TimeCode _message;
125 uint64_t _stepNumerator = 0;
126 uint64_t _stepDenominator = 0;
127 uint64_t _remainder = 0;
128 uint32_t _next = 0;
129 uint8_t _piece = 0;
130 bool _running = false;
131};

143class MtcReceiver : public Device {
144public:
146 void Process(const Protocol::Packet& packet) {
147 if (packet.IsSystem(Protocol::SystemStatus::TimeCode)) {
148 QuarterFrame(packet.TimeCode());
149 } else if (packet.Type() == Protocol::MessageType::Data64) {
150 SysEx(packet);
151 }
152 Emit(packet);
153 }
154
156 Protocol::TimeCode Time() const noexcept { return _time; }
157
158private:
159 void QuarterFrame(uint8_t data) {
160 const uint8_t piece = static_cast<uint8_t>(data >> 4);
161 if (piece != _expected) {
162 _expected = 0;
163 if (piece != 0) {
164 return;
165 }
166 }
167 _nibbles[piece] = static_cast<uint8_t>(data & 0x0Fu);
168 _expected = static_cast<uint8_t>(piece + 1);
169 if (piece == 7) {
170 _expected = 0;
171 const Protocol::TimeCode decoded = Protocol::TimeCode::FromQuarterFrames(_nibbles);
172 if (decoded.IsValid()) {
173 _time = decoded.Next().Next();
174 }
175 }
176 }
177
178 void SysEx(const Protocol::Packet& packet) {
179 const Protocol::SysExStatus status = packet.SysExStatus();
180 if (status == Protocol::SysExStatus::Start || status == Protocol::SysExStatus::Complete) {
181 _sysExSize = 0;
182 }
183 for (uint8_t i = 0; i < packet.SysExSize(); ++i) {
184 if (_sysExSize < FullFrameLength) {
185 _sysEx[_sysExSize] = packet.SysExByte(i);
186 }
187 if (_sysExSize < 0xFF) {
188 ++_sysExSize;
189 }
190 }
191 if ((status == Protocol::SysExStatus::End || status == Protocol::SysExStatus::Complete) &&
192 _sysExSize == FullFrameLength && _sysEx[0] == 0x7F && _sysEx[2] == 0x01 && _sysEx[3] == 0x01) {
193 const Protocol::TimeCode time = Protocol::TimeCode::From(
194 _sysEx[4] & 0x1F, _sysEx[5], _sysEx[6], _sysEx[7],
195 static_cast<Protocol::TimeCodeRate>((_sysEx[4] >> 5) & 3u));
196 if (time.IsValid()) {
197 _time = time;
198 _expected = 0;
199 }
200 }
201 }
202
203 static constexpr uint8_t FullFrameLength = 8;
204
205 Protocol::TimeCode _time;
206 uint8_t _nibbles[8] = {};
207 uint8_t _expected = 0;
208 uint8_t _sysEx[FullFrameLength] = {};
209 uint8_t _sysExSize = 0;
210};

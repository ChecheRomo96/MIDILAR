34inline Midi::Message DecodeMidi1(const Packet& packet) noexcept {
35 Midi::Message result;
36 if (!packet.IsValid() || (packet.Type() != MessageType::Midi1ChannelVoice &&
37 packet.Type() != MessageType::System)) return result;
38 const uint8_t status = static_cast<uint8_t>(packet.Word(0) >> 16);
39 if ((packet.Type() == MessageType::System) != (status >= 0xF0)) return result;
40 const uint8_t bytes[3] = {status, static_cast<uint8_t>(packet.Word(0) >> 8),
41 static_cast<uint8_t>(packet.Word(0))};
42 // Try exact complete-message lengths; SetRawData validates the status.
43 for (size_t size = 1; size <= 3; ++size) {
44 if ((size < 3 && bytes[2] != 0) || (size < 2 && bytes[1] != 0)) continue;
45 result.SetRawData(bytes, size);
46 if (result.IsValid()) return result;
47 }
48 return result;
49}
